# Set to steady colors:
asusctl aura static -c ffffff    # White
asusctl aura static -c ff0000    # Red
asusctl aura static -c 00ff00    # Green
asusctl aura static -c 0000ff    # Blue
asusctl aura static -c ff6600    # Orange

# Other modes:
asusctl aura breathe -c ffffff   # Breathing effect
asusctl aura rainbow            # Rainbow cycling (current mode)
asusctl aura off                # Turn off RGB

# Check current status:
asusctl aura
